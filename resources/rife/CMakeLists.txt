cmake_minimum_required(VERSION 3.16)
project(fold_rife LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NCNN_SYSTEM_GLSLANG OFF CACHE BOOL "" FORCE)
add_subdirectory(${RIFE_SOURCE}/src/ncnn ncnn)
find_package(Vulkan REQUIRED)
set(SHADERS)
foreach(NAME warp warp_pack4 warp_pack8)
    set(HEADER ${CMAKE_CURRENT_BINARY_DIR}/${NAME}.comp.hex.h)
    add_custom_command(OUTPUT ${HEADER}
        COMMAND ${CMAKE_COMMAND} -DSHADER_SRC=${RIFE_SOURCE}/src/${NAME}.comp -DSHADER_COMP_HEADER=${HEADER} -P ${RIFE_SOURCE}/src/generate_shader_comp_header.cmake
        DEPENDS ${RIFE_SOURCE}/src/${NAME}.comp)
    list(APPEND SHADERS ${HEADER})
endforeach()
add_library(fold_rife SHARED adapter.cpp ${RIFE_SOURCE}/src/warp.cpp ${SHADERS})
target_include_directories(fold_rife PRIVATE ${RIFE_SOURCE}/src ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(fold_rife PRIVATE ncnn Vulkan::Vulkan)
set_target_properties(fold_rife PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS fold_rife LIBRARY DESTINATION lib/fold)
